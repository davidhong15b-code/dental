cmake_minimum_required(VERSION 3.8)
project(dental_perception)

find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(dental_interfaces REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d aruco videoio)

foreach(target IN ITEMS perception_node model_localizer_node)
  add_executable(${target} src/${target}.cpp)
  target_compile_features(${target} PUBLIC cxx_std_17)
  target_include_directories(${target} PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(${target} ${OpenCV_LIBS})
  ament_target_dependencies(${target}
    rclcpp rclcpp_action ament_index_cpp sensor_msgs geometry_msgs std_msgs
    dental_interfaces cv_bridge tf2 tf2_ros tf2_geometry_msgs)
endforeach()
# The camera mode uses the RealSense C++ SDK when installed. Image mode works without it.
find_package(realsense2 QUIET CONFIG)
if(realsense2_FOUND)
  if(TARGET realsense2::realsense2)
    target_link_libraries(perception_node realsense2::realsense2)
  elseif(TARGET realsense2)
    target_link_libraries(perception_node realsense2)
  elseif(DEFINED realsense2_LIBRARY)
    target_link_libraries(perception_node ${realsense2_LIBRARY})
  else()
    message(FATAL_ERROR "librealsense2 found but no library target or path is defined")
  endif()
  target_compile_definitions(perception_node PRIVATE DENTAL_HAVE_REALSENSE=1)
  message(STATUS "dental_perception: RealSense camera backend ENABLED")
else()
  message(STATUS "dental_perception: librealsense2 not found; image mode available")
endif()
install(TARGETS perception_node model_localizer_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch samples DESTINATION share/${PROJECT_NAME})
ament_package()
